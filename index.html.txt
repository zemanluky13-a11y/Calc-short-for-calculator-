<script>
    const display = document.getElementById('display');
    const history = document.getElementById('history');
    
    let currentInput = '0';
    let activeOp = null;
    let prevValue = null;
    let resetOnNextKey = false;

    function updateDisplay() {
        display.innerText = currentInput;
    }

    function inputNum(num) {
        if (currentInput === '0' || resetOnNextKey) {
            currentInput = num;
            resetOnNextKey = false;
        } else {
            if (num === '.' && currentInput.includes('.')) return;
            currentInput += num;
        }
        updateDisplay();
    }

    function inputOp(op) {
        prevValue = parseFloat(currentInput);
        activeOp = op;
        history.innerText = `${prevValue} ${op === '*' ? '×' : op === '/' ? '÷' : op}`;
        resetOnNextKey = true;
    }

    function inputFn(fn) {
        let val = parseFloat(currentInput);
        let result = 0;
        
        switch(fn) {
            case 'sin': result = Math.sin(val * Math.PI / 180); break;
            case 'cos': result = Math.cos(val * Math.PI / 180); break;
            case 'tan': result = Math.tan(val * Math.PI / 180); break;
            case 'sqrt': result = Math.sqrt(val); break;
            case 'log': result = Math.log10(val); break;
            case 'pow': 
                prevValue = val;
                activeOp = '^';
                history.innerText = `${prevValue} ^`;
                resetOnNextKey = true;
                return;
        }
        
        history.innerText = `${fn}(${val})`;
        currentInput = Number(result.toFixed(8)).toString();
        resetOnNextKey = true;
        updateDisplay();
    }

    function calculate() {
        if (!activeOp || prevValue === null) return;
        let current = parseFloat(currentInput);
        let result = 0;

        switch(activeOp) {
            case '+': result = prevValue + current; break;
            case '-': result = prevValue - current; break;
            case '*': result = prevValue * current; break;
            case '/': result = current !== 0 ? prevValue / current : 'Chyba'; break;
            case '^': result = Math.pow(prevValue, current); break;
        }

        history.innerText += ` ${current} =`;
        currentInput = Number(result.toFixed(8)).toString();
        activeOp = null;
        prevValue = null;
        resetOnNextKey = true;
        updateDisplay();
    }

    function clearAll() {
        currentInput = '0';
        activeOp = null;
        prevValue = null;
        history.innerText = '';
        updateDisplay();
    }

    function deleteLast() {
        if (currentInput.length > 1) {
            currentInput = currentInput.slice(0, -1);
        } else {
            currentInput = '0';
        }
        updateDisplay();
    }
</script>
